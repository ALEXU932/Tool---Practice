users
Column	Type	Constraints	Description
user_id	INT	PRIMARY KEY, AUTO_INCREMENT	Unique user ID
name	VARCHAR(100)	NOT NULL	Full name
email	VARCHAR(100)	UNIQUE, NOT NULL	Email address
password	VARCHAR(255)	NOT NULL	Hashed password
role	ENUM('admin','vendor','client')	NOT NULL	User role
phone	VARCHAR(20)		Contact number
created_at	DATETIME	DEFAULT CURRENT_TIMESTAMP	Account creation date
updated_at	DATETIME	DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP	Last update

products
Column	Type	Constraints	Description
product_id	INT	PRIMARY KEY, AUTO_INCREMENT	Unique product ID
name	VARCHAR(200)	NOT NULL	Product name
description	TEXT		Product description
price	DECIMAL(10,2)	NOT NULL	Price in ETB
stock	INT	DEFAULT 0	Available quantity
category_id	INT	FOREIGN KEY (category_id) REFERENCES categories(category_id)	Category
vendor_id	INT	FOREIGN KEY (vendor_id) REFERENCES users(user_id)	Vendor who added
image_url	VARCHAR(500)		Product image
created_at	DATETIME	DEFAULT CURRENT_TIMESTAMP	Created date
updated_at	DATETIME	DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP	Last update

categories
Column	Type	Constraints	Description
category_id	INT	PRIMARY KEY, AUTO_INCREMENT	Unique category ID
name	VARCHAR(100)	NOT NULL, UNIQUE	Category name
parent_id	INT	FOREIGN KEY (parent_id) REFERENCES categories(category_id)	Parent category (for nesting)
created_at	DATETIME	DEFAULT CURRENT_TIMESTAMP	Created date

orders
Column	Type	Constraints	Description
order_id	INT	PRIMARY KEY, AUTO_INCREMENT	Unique order ID
user_id	INT	FOREIGN KEY (user_id) REFERENCES users(user_id)	Customer
total_amount	DECIMAL(10,2)	NOT NULL	Total order cost
status	ENUM('pending','processing','shipped','delivered',      'cancelled')	DEFAULT 'pending'	Order status
shipping_address	TEXT	NOT NULL	Delivery address
created_at	DATETIME	DEFAULT CURRENT_TIMESTAMP	Order date
updated_at	DATETIME	DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP	Last update

order_items
Column	Type	Constraints	Description
order_item_id	INT	PRIMARY KEY, AUTO_INCREMENT	Unique item ID
order_id	INT	FOREIGN KEY (order_id) REFERENCES orders(order_id)	Order reference
product_id	INT	FOREIGN KEY (product_id) REFERENCES products(product_id)	Product
quantity	INT	NOT NULL	Quantity ordered
price	DECIMAL(10,2)	NOT NULL	Price at time of order

shopping_cart
Column	Type	Constraints	Description
cart_id	INT	PRIMARY KEY, AUTO_INCREMENT	Unique cart ID
user_id	INT	FOREIGN KEY (user_id) REFERENCES users(user_id)	Cart owner
created_at	DATETIME	DEFAULT CURRENT_TIMESTAMP	Cart creation date

cart_items
Column	Type	Constraints	Description
cart_item_id	INT	PRIMARY KEY, AUTO_INCREMENT	Unique cart item ID
cart_id	INT	FOREIGN KEY (cart_id) REFERENCES shopping_cart(cart_id)	Cart reference
product_id	INT	FOREIGN KEY (product_id) REFERENCES products(product_id)	Product in cart
quantity	INT	NOT NULL	Quantity

payments
Column	Type	Constraints	Description
payment_id	INT	PRIMARY KEY, AUTO_INCREMENT	Unique payment ID
order_id	INT	FOREIGN KEY (order_id) REFERENCES orders(order_id)	Order reference
payment_method	VARCHAR(50)	NOT NULL	e.g., Chapa, COD
amount	DECIMAL(10,2)	NOT NULL	Amount paid
status	ENUM('pending','completed','failed','refunded')	DEFAULT 'pending'	Payment status
transaction_id	VARCHAR(100)		Gateway transaction ID
created_at	DATETIME	DEFAULT CURRENT_TIMESTAMP	Payment date

reviews
Column	Type	Constraints	Description
review_id	INT	PRIMARY KEY, AUTO_INCREMENT	Unique review ID
user_id	INT	FOREIGN KEY (user_id) REFERENCES users(user_id)	Reviewer
product_id	INT	FOREIGN KEY (product_id) REFERENCES products(product_id)	Product reviewed
rating	INT	CHECK (1 <= rating <= 5)	Rating 1–5
comment	TEXT		Review text
created_at	DATETIME	DEFAULT CURRENT_TIMESTAMP	Review date
